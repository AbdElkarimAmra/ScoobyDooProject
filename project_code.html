# keep only valid (non-missing) IMDB and recode variables
model_data <- scoobydoo %>%
  filter(imdb != "NULL") %>%
  mutate(
    imdb = as.numeric(imdb),
    batman = as.numeric(batman),
    blue_falcon = as.numeric(blue_falcon),
    hex_girls = as.numeric(hex_girls),

model <- lm(imdb ~ batman + blue_falcon + hex_girls + scooby_dum + scrappy_doo, data = model_data)
